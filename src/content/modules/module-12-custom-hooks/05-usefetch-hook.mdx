# useFetch Hook

A custom hook encapsulating the network request lifecycle with cancellation support.

### Key Concepts & Rules:

- Returns `{ data, isLoading, error, refetch }`.

### Code Example: Production useFetch Custom Hook

```tsx
export function useFetch<T>(url: string) {
  const [data, setData] = useState<T | null>(null);
  const [isLoading, setIsLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    let ignore = false;
    setIsLoading(true);

    fetch(url)
      .then(res => res.json())
      .then(result => {
        if (!ignore) {
          setData(result);
          setIsLoading(false);
        }
      })
      .catch(err => {
        if (!ignore) {
          setError(err.message);
          setIsLoading(false);
        }
      });

    return () => { ignore = true; };
  }, [url]);

  return { data, isLoading, error };
}
```
