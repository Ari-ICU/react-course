# useMutation Hook

ខុសប្លែកពី Query ដែលដំណើរការដោយស្វ័យប្រវត្តិតាម Lifecycle, Mutation ត្រូវបានបញ្ឆេះឡើងតាមរយៈសកម្មភាពផ្ទាល់របស់អ្នកប្រើប្រាស់ (User Action) ដូចជាការចុចប៊ូតុង Submit Form (POST), កែសម្រួលព័ត៌មាន (PUT/PATCH), ឬលុបទិន្នន័យ (DELETE)។

### Key Concepts & Rules:

- ប្រើប្រាស់ `useMutation({ mutationFn: ... })`។
- ដំណើរការ Mutation តាមរយៈអនុគមន៍ `mutation.mutate(payload)`។
- ប្រើប្រាស់ Lifecycle Callbacks: `onSuccess`, `onError`, `onSettled`។
- ហៅ `queryClient.invalidateQueries` ក្នុង `onSuccess` ដើម្បីធ្វើបច្ចុប្បន្នភាពទិន្នន័យតារាង។

### Code Example: useMutation with Cache Invalidation

```jsx
import { useMutation, useQueryClient } from '@tanstack/react-query';
import { studentService } from '@/services/studentService';

export function AddStudentButton() {
  const queryClient = useQueryClient();

  const mutation = useMutation({
    mutationFn: studentService.create,
    onSuccess: () => {
      // បង្ខំឱ្យបញ្ជីសិស្ស Fetch ឡើងវិញដោយស្វ័យប្រវត្តិ!
      queryClient.invalidateQueries({ queryKey: ['students'] });
      alert('បានបន្ថែមសិស្សដោយជោគជ័យ!');
    },
    onError: (error: any) => {
      alert(`បរាជ័យ៖ ${error.message}`);
    },
  });

  return (
    <button 
      disabled={mutation.isPending}
      onClick={() => mutation.mutate({ name: 'សុភា', gpa: 3.8 })}
    >
      {mutation.isPending ? 'កំពុងរក្សាទុក...' : 'បន្ថែមសិស្ស'}
    </button>
  );
}
```
