# Zustand + React 19

នៅក្នុងសម័យកាលនៃ React 18 និង React 19 ការបង្ហាញផ្ទាំងទិដ្ឋភាពក្នុងពេលដំណាលគ្នា (Concurrent Rendering) អាចបណ្តាលឱ្យមានបញ្ហា **Visual Tearing** (UI ផ្នែកខ្លះបង្ហាញ State ចាស់ ផ្នែកខ្លះបង្ហាញ State ថ្មីក្នុង Frame តែមួយ) ប្រសិនបើបណ្ណាល័យ State ខាងក្រៅមិនត្រូវបានរចនាត្រឹមត្រូវ។ Zustand បានរួមបញ្ចូល និងពឹងផ្អែកលើ `useSyncExternalStore` API ដោយផ្ទាល់ ដើម្បីលុបបំបាត់បញ្ហានេះ។

### Key Concepts & Rules:

- សុវត្ថិភាព 100% ជាមួយ Concurrent Features នៃ React 19 (Transitions, Suspense)។
- គ្មានបញ្ហា Visual Tearing ឬ State Inconsistency ឡើយ។
- ដំណើរការស៊ីសង្វាក់គ្នាយ៉ាងរលូនជាមួយ Server Components និង Client Components។

### Code Example: Concurrent Safety Guarantee

```jsx
import React, { useTransition } from 'react';
import { create } from 'zustand';

interface SearchStore {
  query: string;
  results: string[];
  setQuery: (q: string) => void;
}

export const useSearchStore = create<SearchStore>((set) => ({
  query: '',
  results: [],
  setQuery: (query) => {
    // Zustand uses useSyncExternalStore under the hood to prevent tearing during concurrent transitions
    const results = ['React 19 Hooks', 'Server Components', 'Actions'].filter(i =>
      i.toLowerCase().includes(query.toLowerCase())
    );
    set({ query, results });
  },
}));

export function React19ConcurrentSearch() {
  const [isPending, startTransition] = useTransition();
  const { query, results, setQuery } = useSearchStore();

  const handleChange = (e: React.ChangeEvent<HTMLInputElement>) => {
    const val = e.target.value;
    // Wrapping high-priority user input in React 19 transition safely with Zustand:
    startTransition(() => {
      setQuery(val);
    });
  };

  return (
    <div className="p-5 bg-slate-900 border border-slate-800 rounded-xl space-y-3">
      <input
        type="text"
        value={query}
        onChange={handleChange}
        placeholder="Concurrent search..."
        className="w-full px-3 py-2 bg-slate-800 border border-slate-700 rounded text-white text-sm"
      />
      {isPending && <p className="text-xs text-blue-400 animate-pulse">Rendering transition...</p>}
      <ul className="text-sm text-slate-300">
        {results.map((r, i) => <li key={i}>• {r}</li>)}
      </ul>
    </div>
  );
}
```
