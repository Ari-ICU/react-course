# Multi-Step Form Wizard

នៅពេលដែល Form មានទិន្នន័យច្រើន ដូចជាដំណើរការចុះឈ្មោះក្រុមហ៊ុន ឬការទិញទំនិញ (Checkout) ការបំបែក Form នោះជាដំណាក់កាលៗ (Wizard: ព័ត៌មានផ្ទាល់ខ្លួន -> អាសយដ្ឋាន -> ការទូទាត់) ជួយសម្រួលដល់ User យ៉ាងខ្លាំង។ អនុគមន៍ `trigger()` របស់ React Hook Form អនុញ្ញាតឱ្យយើង Validate តែ Fields នៃជំហានបច្ចុប្បន្នមុននឹងបន្តទៅមុខ។

### Key Concepts & Rules:

- រក្សាទុកទិន្នន័យទាំងអស់ក្នុង `useForm` តែមួយគត់ ដើម្បីកុំឱ្យបាត់បង់ទិន្នន័យពេលប្តូរជំហាន។
- ប្រើ `trigger(['fieldA', 'fieldB'])` ដើម្បីពិនិត្យសុពលភាពមុននឹងអនុញ្ញាតឱ្យទៅ `step + 1`។
- ហៅ `handleSubmit(onSubmit)` ចុងក្រោយបង្អស់តែនៅជំហានបញ្ចប់ប៉ុណ្ណោះ។

### Code Example: Step Validation with trigger()

```jsx
export function MultiStepWizard() {
  const [step, setStep] = useState(1);
  const { register, trigger, handleSubmit } = useForm<FormFields>();

  const handleNext = async () => {
    let isValid = false;
    
    if (step === 1) {
      // ផ្ទៀងផ្ទាត់តែ Field ក្នុងជំហានទី ១ ប៉ុណ្ណោះ
      isValid = await trigger(['name', 'email']);
    } else if (step === 2) {
      // ផ្ទៀងផ្ទាត់តែ Field ក្នុងជំហានទី ២
      isValid = await trigger(['address', 'city']);
    }

    if (isValid) {
      setStep((s) => s + 1);
    }
  };

  return (
    <div>
      {step === 1 && <StepOne register={register} />}
      {step === 2 && <StepTwo register={register} />}
      {step === 3 && <StepThree register={register} />}
      
      <button type="button" onClick={handleNext}>បន្ទាប់ ➔</button>
    </div>
  );
}
```
